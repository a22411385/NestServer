```typescript
export enum MaterialType {
  POISON_ESSENCE = 'poison_essence',      // 毒精華
  FROST_ESSENCE = 'frost_essence',        // 冰霜精華
  FLAME_ESSENCE = 'flame_essence',        // 火焰精華
  LIGHTNING_ESSENCE = 'lightning_essence',// 閃電精華

  WEAPON_SCROLL = 'weapon_scroll',        // 武器卷軸

}

// 3種稀有度
export enum MaterialRarity {
  COMMON = 'common',    // 普通 (15-20%掉落)
  RARE = 'rare',        // 稀有 (5-8%掉落)
  EPIC = 'epic'         // 史詩 (2-3%掉落)
}

// 3種類別
export enum MaterialCategory {
  ESSENCE = 'essence',  // 精華類 (5種)
  ORE = 'ore',         // 礦石類 (3種)
  SPECIAL = 'special'   // 特殊類 (2種)
}

// 素材配置接口 (14個欄位)
export interface MaterialConfigDefinition {
  id: string;                    // 素材ID (例: poison_essence)
  name: string;                  // 顯示名稱 (例: 毒精華)
  description: string;           // 描述
  rarity: MaterialRarity;        // 稀有度
  category: MaterialCategory;    // 類別
  stackSize: number;             // 堆疊上限 (通常999)
  sellPrice: number;             // 商店售價
  baseDropRate: number;          // 基礎掉落率 (%)
  minDropQuantity: number;       // 最小掉落數量
  maxDropQuantity: number;       // 最大掉落數量
  iconPath: string;              // 圖示路徑
  usedInRecipes: string;         // 用於哪些配方 (逗號分隔)
  dropFromEnemyLevel: number;    // 從哪個等級的敵人開始掉落
  enabled: boolean;              // 是否啟用
}

// 掉落配置接口
export interface MaterialDropConfig {
  materialId: string;
  dropRate: number;
  minQuantity: number;
  maxQuantity: number;
  requiredEnemyLevel: number;
}
```

### 4. 配置管理器擴展 (ConfigManager.ts)

**檔案位置:** `src/Game/Managers/ConfigManager.ts`

添加了6個素材配置查詢方法:

```typescript
import { MaterialConfigDefinition } from '@/Types/Equipment/MaterialTypes';

export class ConfigManager {
    /**
     * 獲取所有素材配置
     */
    public static getMaterialConfigs(): MaterialConfigDefinition[] {
        const cache = this.loadCache();
        return cache.MaterialConfigs || [];
    }

    /**
     * 根據ID獲取素材配置
     */
    public static getMaterialConfigById(materialId: string): MaterialConfigDefinition | null {
        const materials = this.getMaterialConfigs();
        return materials.find(material => material.id === materialId) || null;
    }

    /**
     * 根據稀有度獲取素材列表
     */
    public static getMaterialsByRarity(rarity: string): MaterialConfigDefinition[] {
        const materials = this.getMaterialConfigs();
        return materials.filter(material => material.rarity === rarity && material.enabled);
    }

    /**
     * 根據類別獲取素材列表
     */
    public static getMaterialsByCategory(category: string): MaterialConfigDefinition[] {
        const materials = this.getMaterialConfigs();
        return materials.filter(material => material.category === category && material.enabled);
    }

    /**
     * 根據敵人等級獲取可掉落的素材列表
     */
    public static getMaterialsByEnemyLevel(enemyLevel: number): MaterialConfigDefinition[] {
        const materials = this.getMaterialConfigs();
        return materials.filter(material => 
            material.enabled && 
            material.dropFromEnemyLevel <= enemyLevel
        );
    }

    /**
     * 獲取啟用的素材列表
     */
    public static getEnabledMaterials(): MaterialConfigDefinition[] {
        const materials = this.getMaterialConfigs();
        return materials.filter(material => material.enabled);
    }
}
```

## 🔍 使用範例

### 在 DropSystem 中使用素材配置

```typescript
import { ConfigManager } from '@/Game/Managers/ConfigManager';
import { MaterialType } from '@/Types/Equipment/MaterialTypes';

export class DropSystem {
    /**
     * 根據敵人等級產生素材掉落
     */
    public generateMaterialDrops(enemyLevel: number, enemyX: number, enemyY: number): ServerItem[] {
        const drops: ServerItem[] = [];
        
        // 獲取該等級可掉落的素材列表
        const availableMaterials = ConfigManager.getMaterialsByEnemyLevel(enemyLevel);
        
        for (const materialConfig of availableMaterials) {
            // 檢查是否成功掉落
            const dropChance = Math.random() * 100;
            if (dropChance <= materialConfig.baseDropRate) {
                // 隨機生成掉落數量
                const quantity = Math.floor(
                    Math.random() * (materialConfig.maxDropQuantity - materialConfig.minDropQuantity + 1)
                ) + materialConfig.minDropQuantity;
                
                // 創建素材物品
                const material = ServerItem.createMaterial(
                    enemyX,
                    enemyY,
                    materialConfig.id,
                    materialConfig.name,
                    quantity
                );
                
                drops.push(material);
            }
        }
        
        return drops;
    }
}
```

### 在 ItemPickupSystem 中使用

```typescript
import { ConfigManager } from '@/Game/Managers/ConfigManager';

export class ItemPickupSystem {
    /**
     * 拾取素材並加入角色的素材庫存
     */
    public pickupMaterial(hero: HeroSchema, material: ServerItem): boolean {
        const materialConfig = ConfigManager.getMaterialConfigById(material.itemId);
        
        if (!materialConfig) {
            console.error(`素材配置不存在: ${material.itemId}`);
            return false;
        }
        
        // 計算當前擁有的數量
        const currentQuantity = hero.materials.get(material.itemId) || 0;
        
        // 檢查是否超過堆疊上限
        if (currentQuantity + material.quantity > materialConfig.stackSize) {
            console.warn(`素材 ${materialConfig.name} 已達堆疊上限`);
            return false;
        }
        
        // 更新素材數量 (使用 MapSchema)
        hero.materials.set(material.itemId, currentQuantity + material.quantity);
        
        return true;
    }
}
```

### 在 CraftingSystem 中查詢素材

```typescript
import { ConfigManager } from '@/Game/Managers/ConfigManager';
import { MaterialCategory } from '@/Types/Equipment/MaterialTypes';

export class CraftingSystem {
    /**
     * 獲取所有精華類素材
     */
    public getEssences(): MaterialConfigDefinition[] {
        return ConfigManager.getMaterialsByCategory(MaterialCategory.ESSENCE);
    }
    
    /**
     * 獲取所有稀有素材
     */
    public getRareMaterials(): MaterialConfigDefinition[] {
        return ConfigManager.getMaterialsByRarity('rare');
    }
    
    /**
     * 檢查玩家是否有足夠的素材
     */
    public hasEnoughMaterials(
        hero: HeroSchema, 
        requiredMaterials: Record<string, number>
    ): boolean {
        for (const [materialId, requiredQuantity] of Object.entries(requiredMaterials)) {
            const currentQuantity = hero.materials.get(materialId) || 0;
            if (currentQuantity < requiredQuantity) {
                return false;
            }
        }
        return true;
    }
}
```
